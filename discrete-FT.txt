%Objective: To compute the Discrete Time Fourier Transform of signal.

%difference btween FT and DTFT
%

clc
clear
close

%scale factor should be varied and observed
N1 = 10; % time duration of the pulse being 1

% Signal 2
% time duration for which signal is being generated
n = -N1:1:N1;
x=zeros(size(n));

% generating the rectangular pulse x(t)
x(n >= -N1 & n<0) = -1.0;
x(n >= 0 & n<=N1) =  1.0;
 
% % Signal 3
% n = -3.0*N1:1:3.0*N1;
% x(n >= -N1) = n(n >= -N1);
% x(n > N1) = 0.0;

% Plot of the signal
figure;
subplot(3,1,1);
stem(n,x);
%axis([min(n) max(n) 1.1*min(x) 1.1*max(x)]);
grid on
title('Discrete Time Signal')
xlabel('n')
ylabel('x[n]')

omega = -N1:0.01:N1;
X_omega = zeros(size(omega));

for ind = 1:length(n)
    X_omega = X_omega + x(ind)*exp(-1i*omega*n(ind));
end

% Plot of the spectral content
subplot(3,1,2);
plot(omega,abs(X_omega));
grid on
title('DTFT magnitude')
xlabel('\omega')
ylabel('X(\omega)')

%set(gca,'xtick',[-5*pi:pi:5*pi])
%set(gca,'xticklabels',{'-5\pi','-4\pi','-3\pi','-2\pi','-\pi','0','\pi','2\pi','3\pi','4\pi','5\pi'})

subplot(3,1,3);
plot(omega,phase(X_omega));
%axis([min(omega) max(omega) -2*pi 2*pi]);

grid on
title('DTFT phase')
xlabel('\omega')
ylabel('X(\omega)')
