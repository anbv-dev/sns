%Objective: To study the shifting, time reversal and scaling of a given continuous time domain signal

clc;
clear;
close all;

% the limits of the time for the signal are defined here
t=0:0.01:4*pi;
x=sin(t);

% The plot of original signal with the limits of axes
subplot(5,1,1);
plot(t,x);
grid on
title('original signal')
xlabel('t');
ylabel('x');

% The plot of shifted signal with the limits of axes
subplot(5,1,2);
tnew = t + 3;
plot(tnew ,x)
grid on
title('shifted signal')
xlabel('t')
ylabel('x(t-3)')

% The plot of scaled signal with the limits of axes
subplot(5,1,3);
tnew = t/2;
plot(tnew,x)
grid on
title('scaled signal')
xlabel('t')
ylabel('x(2t)')

% The plot of time reversed signal with the limits of axes
subplot(5,1,4);
tnew = -t;
plot(tnew,x)
grid on
title('time reversed signal')
xlabel('t')
ylabel('x(-t)')

% The plot of three transformation of the signal with the limits of axes
subplot(5,1,5);
tnew = -t/2 - 3/2 ;
plot(tnew,x)
grid on
title('all combined')
xlabel('t')
ylabel('x(-2t - 3)')
